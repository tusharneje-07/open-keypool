open_keypool

open_keypool — a minimal Python library for pooling and rotating API keys.

Avoids HTTP 429 rate-limit errors by cycling through a pool of keys with cooldown and disablement support. Provide a list of keys (or pull them from Doppler), choose a rotation strategy (round-robin or least-recently-used), and the pool handles cooldown on rate-limit responses and permanent disablement on invalid keys — all thread-safe.

Install

pip install open-keypool

Quickstart — Local keys array

from open_keypool import KeyPool, AllKeysExhaustedError

pool = KeyPool(keys=["sk-key1", "sk-key2", "sk-key3"], strategy="round_robin")

for attempt in range(pool.max_retries):
    key = pool.get_key()
    response = call_your_api(key)
    if response.status_code == 429:
        retry_after = float(response.headers.get("Retry-After", 0))
        pool.mark_rate_limited(key, retry_after=retry_after or None)
    elif response.status_code in (401, 403):
        pool.mark_invalid(key)
    else:
        pool.mark_success(key)
        break

 90class KeyPool:
 91    """A thread-safe pool of API keys with cooldown and rotation strategies.
 92
 93    Manages a collection of API keys, cycling through them using a configurable
 94    strategy (round-robin or least-recently-used). Keys that receive HTTP 429
 95    responses can be placed on cooldown and will automatically recover after
 96    the cooldown period. Permanently invalid keys can be explicitly disabled.
 97
 98    Parameters
 99    ----------
100    keys : list[str]
101        Initial list of API key strings. At least one key is required.
102    max_retries : int, optional
103        Maximum number of retries the caller should attempt per operation.
104        Stored as ``self.max_retries`` for the caller's reference; the library
105        does not perform automatic retries. Default is 3.
106    cooldown_seconds : int, optional
107        Number of seconds a key stays in COOLDOWN after being rate-limited
108        before it becomes ACTIVE again. Default is 60.
109    strategy : str, optional
110        Rotation strategy. ``"round_robin"`` cycles through keys in insertion
111        order. ``"lru"`` selects the key with the oldest ``last_used``
112        timestamp. Default is ``"round_robin"``.
113
114    Raises
115    ------
116    ValueError
117        If *keys* is empty or ``None``.
118
119    Examples
120    --------
121    >>> pool = KeyPool(["key-a", "key-b", "key-c"], strategy="round_robin")
122    >>> pool.get_key()
123    >>> pool.mark_success(pool.get_key())
124    """

151    def _find_record(self, key: str) -> _KeyRecord | None:
152        """Return the ``_KeyRecord`` for *key*, or ``None`` if not found."""
153        for rec in self._records:
154            if rec.key == key:
155                return rec
156        return None
157
158    def _recover_cooldown_keys(self) -> None:
159        """Flip COOLDOWN keys whose cooldown has expired back to ACTIVE."""
160        now = time.monotonic()
161        for rec in self._records:
162            if rec.state == KeyState.COOLDOWN and rec.cooldown_until is not None and now >= rec.cooldown_until:
163                rec.state = KeyState.ACTIVE
164                rec.cooldown_until = None
165
166    def _active_records(self) -> list[_KeyRecord]:
167        """Return all currently-ACTIVE records."""
168        return [r for r in self._records if r.state == KeyState.ACTIVE]

171    def from_doppler(
172        cls,
173        token: str,
174        project: str,
175        config: str,
176        key_prefix: str | None = None,
177        max_retries: int = 3,
178        cooldown_seconds: int = 60,
179        strategy: str = "round_robin",
180        force_refresh: bool = False,
181    ) -> KeyPool:
182        """Create a ``KeyPool`` by fetching API keys from Doppler.
183
184        Calls the Doppler secrets-download REST endpoint and populates the
185        pool with secret values that match *key_prefix* (if given). Results
186        are cached in an in-memory, module-level ``TTLCache`` (1 hour TTL)
187        keyed by ``(project, config, key_prefix)`` so that repeated calls
188        within the same process avoid redundant network requests.
189
190        The cache is purely in-memory and empties naturally on every fresh
191        process start — it is never persisted to disk.
192
193        Parameters
194        ----------
195        token : str
196            Doppler service-token for bearer authentication
197            (e.g. ``"dp.st.YOUR_SERVICE_TOKEN"``).
198        project : str
199            Doppler project name.
200        config : str
201            Doppler config/environment name (e.g. ``"dev"``, ``"prd"``).
202        key_prefix : str | None, optional
203            If provided, only secrets whose name starts with this string are
204            included as keys. ``None`` (default) includes all secrets.
205        max_retries : int, optional
206            Passed through to the ``KeyPool`` constructor. Default is 3.
207        cooldown_seconds : int, optional
208            Passed through to the ``KeyPool`` constructor. Default is 60.
209        strategy : str, optional
210            Passed through to the ``KeyPool`` constructor.
211            Default is ``"round_robin"``.
212        force_refresh : bool, optional
213            If ``True``, bypass the cache and re-fetch from Doppler even when
214            a valid cache entry exists. Default is ``False``.
215
216        Returns
217        -------
218        KeyPool
219            A new ``KeyPool`` instance populated with the fetched keys.
220
221        Raises
222        ------
223        RuntimeError
224            If the Doppler API call fails (non-2xx status) or returns zero
225            keys — the cache is **not** populated on failure.
226
227        Examples
228        --------
229        >>> import os
230        >>> pool = KeyPool.from_doppler(
231        ...     token=os.getenv("DOPPLER_TOKEN", "dp.st.YOUR_SERVICE_TOKEN"),
232        ...     project="refactor-ai",
233        ...     config="dev",
234        ...     key_prefix="API_KEY_",
235        ... )
236        """
237        cache_key = (project, config, key_prefix)
238
239        if not force_refresh:
240            cached = _DOPPLER_CACHE.get(cache_key)
241            if cached is not None:
242                return cls(
243                    keys=list(cached),
244                    max_retries=max_retries,
245                    cooldown_seconds=cooldown_seconds,
246                    strategy=strategy,
247                )
248
249        try:
250            response = httpx.get(
251                _DOPPLER_DOWNLOAD_URL,
252                params={"project": project, "config": config},
253                headers={"Authorization": f"Bearer {token}"},
254            )
255            response.raise_for_status()
256            data = response.json()
257        except httpx.HTTPError as exc:
258            raise RuntimeError(
259                f"Doppler API request failed: {exc.__class__.__name__}"
260            ) from exc
261
262        secrets: dict[str, str] = data.get("secrets", {})
263        fetched_keys: list[str] = []
264        for name, secret_data in secrets.items():
265            if isinstance(secret_data, dict):
266                raw = secret_data.get("raw") or secret_data.get("computed", "")
267            else:
268                raw = str(secret_data)
269            if key_prefix is None or name.startswith(key_prefix):
270                fetched_keys.append(raw)
271
272        if not fetched_keys:
273            raise RuntimeError(
274                f"Doppler returned zero keys for project='{project}', "
275                f"config='{config}', key_prefix={key_prefix!r}"
276            )
277
278        _DOPPLER_CACHE[cache_key] = tuple(fetched_keys)
279
280        return cls(
281            keys=fetched_keys,
282            max_retries=max_retries,
283            cooldown_seconds=cooldown_seconds,
284            strategy=strategy,
285        )

330    def get_key(self) -> str:
331        """Return the next available ACTIVE key according to the pool's strategy.
332
333        Before selecting, any COOLDOWN key whose cooldown period has expired
334        is automatically flipped back to ACTIVE.
335
336        **Round-robin** (``strategy="round_robin"``): iterates through keys
337        in insertion order, maintaining an internal cursor that wraps around.
338
339        **LRU** (``strategy="lru"``): picks the ACTIVE key with the oldest
340        ``last_used`` timestamp and updates it to now upon selection.
341
342        Returns
343        -------
344        str
345            An ACTIVE API key.
346
347        Raises
348        ------
349        AllKeysExhaustedError
350            If no ACTIVE key exists in the pool. The message includes the
351            soonest recovery time in seconds when at least one key is in
352            COOLDOWN, otherwise it says all keys are disabled.
353
354        Examples
355        --------
356        >>> pool = KeyPool(["key-a", "key-b"])
357        >>> key = pool.get_key()
358        >>> pool.mark_success(key)
359        """
360        with self._lock:
361            self._recover_cooldown_keys()
362            active = self._active_records()
363
364            if not active:
365                cooldown_records = [r for r in self._records if r.state == KeyState.COOLDOWN]
366                if cooldown_records:
367                    now = time.monotonic()
368                    soonest = min(
369                        (r.cooldown_until - now for r in cooldown_records if r.cooldown_until is not None),
370                        default=None,
371                    )
372                    if soonest is not None and soonest > 0:
373                        raise AllKeysExhaustedError(
374                            f"No active keys available. Recovery in {soonest:.1f}s "
375                            f"({self._cooldown_summary(cooldown_records, now)})."
376                        )
377                    else:
378                        raise AllKeysExhaustedError(
379                            "No active keys available. All keys are in cooldown or disabled."
380                        )
381                raise AllKeysExhaustedError("No active keys available. All keys are disabled.")
382
383            if self.strategy == "round_robin":
384                self._round_robin_index %= len(active)
385                rec = active[self._round_robin_index]
386                self._round_robin_index += 1
387            else:  # lru
388                rec = min(active, key=lambda r: r.last_used)
389                rec.last_used = time.monotonic()
390
391        return rec.key

393    def handle_response(
394        self,
395        key: str,
396        status_code: int,
397        headers: dict[str, str] | None = None,
398        body: dict | str | None = None,
399    ) -> KeyState:
400        """Feed an HTTP response to the pool — it decides what to do with the key.
401
402        Introspects the status code and response body and automatically:
403
404        - On **2xx**: marks the key as successful (``mark_success``).
405        - On **429** or **413**, or when the response body contains
406          ``error.code == "rate_limit_exceeded"``: places the key on
407          COOLDOWN using ``Retry-After`` if present, otherwise the pool's
408          ``cooldown_seconds``.
409        - On **401** or **403**: permanently disables the key
410          (``mark_invalid``).
411        - On **5xx**: places the key on COOLDOWN (transient server error).
412
413        All relevant details (status code, error code, error message) are
414        stored on the key record and surfaced in ``status()``.
415
416        Parameters
417        ----------
418        key : str
419            The API key that was used for the request.
420        status_code : int
421            HTTP status code from the response.
422        headers : dict[str, str] | None, optional
423            Response headers (used to extract ``Retry-After``).
424        body : dict | str | None, optional
425            Parsed JSON body (``dict``) or raw response text (``str``).
426
427        Returns
428        -------
429        KeyState
430            The new state of the key after processing.
431
432        Examples
433        --------
434        >>> pool = KeyPool(["key-a", "key-b"])
435        >>> k = pool.get_key()
436        >>> # Successful call:
437        >>> pool.handle_response(k, 200, body={"choices": [...]})
438        <KeyState.ACTIVE: 'active'>
439        >>> # Rate-limit (Groq-style in-body):
440        >>> pool.handle_response(k, 200, body={"error": {"code": "rate_limit_exceeded", "message": "TPM limit"}})
441        <KeyState.COOLDOWN: 'cooldown'>
442        >>> # Re-raise to get a fresh key on cooldown:
443        >>> k2 = pool.get_key()
444        """
445        headers = headers or {}
446
447        # ── parse body for error details ──
448        error_code = str(status_code)
449        error_message = ""
450        if isinstance(body, dict):
451            err = body.get("error", {})
452            if isinstance(err, dict):
453                if err.get("code") == "rate_limit_exceeded":
454                    error_code = "rate_limit_exceeded"
455                error_message = err.get("message", "")
456            elif isinstance(err, str):
457                error_message = err
458        elif isinstance(body, str):
459            error_message = body[:200]
460
461        with self._lock:
462            rec = self._find_record(key)
463            if rec is None:
464                return KeyState.DISABLED  # key doesn't exist, nothing to do
465
466            rec.last_status_code = status_code
467
468            # ── 2xx success ──
469            if 200 <= status_code < 300:
470                rec.failure_count = 0
471                rec.state = KeyState.ACTIVE
472                rec.last_error_code = None
473                rec.last_error_message = None
474                return KeyState.ACTIVE
475
476            # ── rate-limit (429, 413, or rate_limit_exceeded in body) ──
477            if status_code in (429, 413) or error_code == "rate_limit_exceeded":
478                ra = headers.get("Retry-After")
479                try:
480                    retry_after = float(ra) if ra else None
481                except (ValueError, TypeError):
482                    retry_after = None
483                rec.state = KeyState.COOLDOWN
484                rec.cooldown_until = time.monotonic() + (
485                    retry_after if retry_after is not None else self.cooldown_seconds
486                )
487                rec.failure_count += 1
488                rec.last_error_code = error_code
489                rec.last_error_message = error_message
490                return KeyState.COOLDOWN
491
492            # ── auth failure (401, 403) ──
493            if status_code in (401, 403):
494                rec.state = KeyState.DISABLED
495                rec.last_error_code = error_code
496                rec.last_error_message = error_message
497                return KeyState.DISABLED
498
499            # ── server error (5xx) — transient, put on cooldown ──
500            if 500 <= status_code < 600:
501                rec.state = KeyState.COOLDOWN
502                rec.cooldown_until = time.monotonic() + self.cooldown_seconds
503                rec.failure_count += 1
504                rec.last_error_code = error_code
505                rec.last_error_message = error_message
506                return KeyState.COOLDOWN
507
508            # ── unknown status — also cooldown ──
509            rec.state = KeyState.COOLDOWN
510            rec.cooldown_until = time.monotonic() + self.cooldown_seconds
511            rec.failure_count += 1
512            rec.last_error_code = error_code
513            rec.last_error_message = error_message
514            return KeyState.COOLDOWN

516    def mark_rate_limited(
517        self,
518        key: str,
519        retry_after: float | None = None,
520        error_code: str | None = None,
521        error_message: str | None = None,
522    ) -> None:
523        """Mark a key as rate-limited (COOLDOWN).
524
525        The key will remain in COOLDOWN for *retry_after* seconds (or the
526        pool's ``cooldown_seconds`` if *retry_after* is ``None``). Its
527        ``failure_count`` is incremented.
528
529        Parameters
530        ----------
531        key : str
532            The API key string.
533        retry_after : float | None, optional
534            Custom cooldown duration in seconds. If ``None``, defaults to
535            ``self.cooldown_seconds``.
536        error_code : str | None, optional
537            Machine-readable code for the last rate-limit error (e.g.
538            ``"rate_limit_exceeded"``, ``"413"``). Stored and surfaced in
539            ``status()``.
540        error_message : str | None, optional
541            Human-readable description of the last rate-limit error.
542            Stored and surfaced in ``status()``.
543
544        Examples
545        --------
546        >>> pool = KeyPool(["key-a"])
547        >>> pool.mark_rate_limited("key-a", retry_after=30)
548        >>> pool.mark_rate_limited("key-a", error_code="rate_limit_exceeded",
549        ...                        error_message="TPM limit 8000 exceeded")
550        """
551        with self._lock:
552            rec = self._find_record(key)
553            if rec is None:
554                return
555            rec.state = KeyState.COOLDOWN
556            rec.cooldown_until = time.monotonic() + (retry_after if retry_after is not None else self.cooldown_seconds)
557            rec.failure_count += 1
558            rec.last_error_code = error_code
559            rec.last_error_message = error_message

595    def mark_success(self, key: str) -> None:
596        """Reset a key's failure count to 0 and keep it ACTIVE.
597
598        Call this after a successful API response to indicate the key is
599        healthy and reset any transient failure tracking.
600
601        Parameters
602        ----------
603        key : str
604            The API key string.
605
606        Examples
607        --------
608        >>> pool = KeyPool(["key-a"])
609        >>> k = pool.get_key()
610        >>> pool.mark_success(k)
611        """
612        with self._lock:
613            rec = self._find_record(key)
614            if rec is None:
615                return
616            rec.failure_count = 0
617            rec.state = KeyState.ACTIVE
618            rec.last_status_code = None
619            rec.last_error_code = None
620            rec.last_error_message = None

622    def status(self) -> dict[str, dict]:
623        """Return a snapshot of every key's state without exposing raw keys.
624
625        Every key value in the returned dictionary is passed through ``mask()``
626        so the caller can safely log or print the result.
627
628        Returns
629        -------
630        dict[str, dict]
631            A mapping of ``{masked_key: {"state": str, "failure_count": int,
632            "cooldown_remaining": float | None, "last_status_code": int | None,
633            "last_error_code": str | None, "last_error_message": str | None}}``
634            for every key in the pool.
635
636        Examples
637        --------
638        >>> pool = KeyPool(["sk-abcdef1234567890"])
639        >>> pool.status()
640        {'sk-abc...7890': {'state': 'active', 'failure_count': 0, 'cooldown_remaining': None, 'last_status_code': None, 'last_error_code': None, 'last_error_message': None}}
641        """
642        with self._lock:
643            result: dict[str, dict] = {}
644            now = time.monotonic()
645            for rec in self._records:
646                if rec.state == KeyState.COOLDOWN and rec.cooldown_until is not None:
647                    remaining = max(0.0, rec.cooldown_until - now)
648                else:
649                    remaining = None
650                result[mask(rec.key)] = {
651                    "state": rec.state.value,
652                    "failure_count": rec.failure_count,
653                    "cooldown_remaining": round(remaining, 1) if remaining is not None else None,
654                    "last_status_code": rec.last_status_code,
655                    "last_error_code": rec.last_error_code,
