open_keypool

open_keypool — a minimal Python library for pooling and rotating API keys.

Avoids HTTP 429 rate-limit errors by cycling through a pool of keys with cooldown and disablement support. Provide a list of keys (or pull them from Doppler, AWS, GCP, .env, or JSON), choose a rotation strategy (round-robin or least-recently-used), and the pool handles cooldown on rate-limit responses and permanent disablement on invalid keys — all thread-safe and async-compatible.

Install

pip install open-keypool

Quickstart — High-level pool.call()

import httpx
from open_keypool import KeyPool

pool = KeyPool(keys=["sk-key1", "sk-key2"], provider="groq")

def fetch_models(key):
    return httpx.get("https://api.groq.com/v1/models", headers={"Authorization": f"Bearer {key}"})

response = pool.call(fetch_models)

Quickstart — AsyncKeyPool

import httpx
import asyncio
from open_keypool import AsyncKeyPool

async def main():
    pool = AsyncKeyPool(keys=["sk-key1", "sk-key2"], provider="openai")

    async def fetch_models(key):
        async with httpx.AsyncClient() as client:
            return await client.get("https://api.openai.com/v1/models", headers={"Authorization": f"Bearer {key}"})

    response = await pool.call(fetch_models)

asyncio.run(main())

 296    def _find_record(self, key: str) -> _KeyRecord | None:
 297        """Return the ``_KeyRecord`` for *key*, or ``None`` if not found."""
 298        for rec in self._records:
 299            if rec.key == key:
 300                return rec
 301        return None
 302
 303    def _recover_cooldown_keys(self) -> None:
 304        """Flip COOLDOWN keys whose cooldown has expired back to ACTIVE."""
 305        now = time.monotonic()
 306        for rec in self._records:
 307            if rec.state == KeyState.COOLDOWN and rec.cooldown_until is not None and now >= rec.cooldown_until:
 308                rec.state = KeyState.ACTIVE
 309                rec.cooldown_until = None
 310                logger.info("key %s recovered", mask(rec.key))
 311
 312    def _active_records(self) -> list[_KeyRecord]:
 313        """Return all currently-ACTIVE records."""
 314        return [r for r in self._records if r.state == KeyState.ACTIVE]

 665    def get_key(self) -> str:
 666        """Return the next available ACTIVE key according to the pool's strategy.
 667
 668        Before selecting, any COOLDOWN key whose cooldown period has expired
 669        is automatically flipped back to ACTIVE.
 670
 671        **Round-robin** (``strategy="round_robin"``): iterates through keys
 672        in insertion order, maintaining an internal cursor that wraps around.
 673
 674        **LRU** (``strategy="lru"``): picks the ACTIVE key with the oldest
 675        ``last_used`` timestamp and updates it to now upon selection.
 676
 677        Returns
 678        -------
 679        str
 680            An ACTIVE API key.
 681
 682        Raises
 683        ------
 684        AllKeysExhaustedError
 685            If no ACTIVE key exists in the pool. The message includes the
 686            soonest recovery time in seconds when at least one key is in
 687            COOLDOWN, otherwise it says all keys are disabled.
 688
 689        Examples
 690        --------
 691        >>> pool = KeyPool(["key-a", "key-b"])
 692        >>> key = pool.get_key()
 693        >>> pool.mark_success(key)
 694        """
 695        with self._lock:
 696            self._recover_cooldown_keys()
 697            active = self._active_records()
 698
 699            if not active:
 700                cooldown_records = [r for r in self._records if r.state == KeyState.COOLDOWN]
 701                if cooldown_records:
 702                    now = time.monotonic()
 703                    soonest = min(
 704                        (r.cooldown_until - now for r in cooldown_records if r.cooldown_until is not None),
 705                        default=None,
 706                    )
 707                    if soonest is not None and soonest > 0:
 708                        raise AllKeysExhaustedError(
 709                            f"No active keys available. Recovery in {soonest:.1f}s "
 710                            f"({self._cooldown_summary(cooldown_records, now)})."
 711                        )
 712                    else:
 713                        raise AllKeysExhaustedError(
 714                            "No active keys available. All keys are in cooldown or disabled."
 715                        )
 716                raise AllKeysExhaustedError("No active keys available. All keys are disabled.")
 717
 718            if self.strategy == "round_robin":
 719                self._round_robin_index %= len(active)
 720                rec = active[self._round_robin_index]
 721                self._round_robin_index += 1
 722            else:  # lru
 723                rec = min(active, key=lambda r: r.last_used)
 724                rec.last_used = time.monotonic()
 725
 726        logger.debug("rotating to key %s", mask(rec.key))
 727        return rec.key

 729    def handle_response(
 730        self,
 731        key: str,
 732        status_code: int,
 733        headers: dict[str, str] | None = None,
 734        body: dict | str | None = None,
 735    ) -> KeyState:
 736        """Feed an HTTP response to the pool — it decides what to do with the key.
 737
 738        Introspects the status code and response body and automatically:
 739
 740        - On **2xx**: marks the key as successful (``mark_success``).
 741        - On **429** or **413**, or when the response body contains
 742          ``error.code == "rate_limit_exceeded"``: places the key on
 743          COOLDOWN using ``Retry-After`` if present, otherwise the pool's
 744          ``cooldown_seconds``.
 745        - On **401** or **403**: permanently disables the key
 746          (``mark_invalid``).
 747        - On **5xx**: places the key on COOLDOWN (transient server error).
 748
 749        All relevant details (status code, error code, error message) are
 750        stored on the key record and surfaced in ``status()``.
 751
 752        Parameters
 753        ----------
 754        key : str
 755            The API key that was used for the request.
 756        status_code : int
 757            HTTP status code from the response.
 758        headers : dict[str, str] | None, optional
 759            Response headers (used to extract ``Retry-After``).
 760        body : dict | str | None, optional
 761            Parsed JSON body (``dict``) or raw response text (``str``).
 762
 763        Returns
 764        -------
 765        KeyState
 766            The new state of the key after processing.
 767
 768        Examples
 769        --------
 770        >>> pool = KeyPool(["key-a", "key-b"])
 771        >>> k = pool.get_key()
 772        >>> # Successful call:
 773        >>> pool.handle_response(k, 200, body={"choices": [...]})
 774        <KeyState.ACTIVE: 'active'>
 775        >>> # Rate-limit (Groq-style in-body):
 776        >>> pool.handle_response(k, 200, body={"error": {"code": "rate_limit_exceeded", "message": "TPM limit"}})
 777        <KeyState.COOLDOWN: 'cooldown'>
 778        >>> # Re-raise to get a fresh key on cooldown:
 779        >>> k2 = pool.get_key()
 780        """
 781    def handle_response(
 782        self,
 783        key: str,
 784        status_code: int,
 785        headers: dict[str, str] | None = None,
 786        body: dict | str | None = None,
 787    ) -> KeyState:
 788        """Feed an HTTP response to the pool — it decides what to do with the key."""
 789        outcome, ra_sec, err_code, err_msg = _process_response_metadata(
 790            self.provider, status_code, headers, body
 791        )
 792
 793        callback: tuple[Callable[..., Any] | None, tuple[Any, ...]] | None = None
 794        with self._lock:
 795            rec = self._find_record(key)
 796            if rec is None:
 797                return KeyState.DISABLED
 798
 799            rec.last_status_code = status_code
 800            err_msg = self._safe_message(err_msg)
 801
 802            if outcome == ResponseOutcome.ACTIVE:
 803                rec.failure_count = 0
 804                rec.state = KeyState.ACTIVE
 805                rec.last_error_code = None if 200 <= status_code < 300 else err_code
 806                rec.last_error_message = None if 200 <= status_code < 300 else err_msg
 807                result = KeyState.ACTIVE
 808            elif outcome == ResponseOutcome.DISABLED:
 809                rec.state = KeyState.DISABLED
 810                rec.last_error_code = err_code
 811                rec.last_error_message = err_msg
 812                callback = (self.on_disabled, (mask(rec.key), err_msg or err_code))
 813                result = KeyState.DISABLED
 814            else:
 815                duration = ra_sec if ra_sec is not None else (
 816                    self.transient_cooldown_seconds if status_code == 408 or status_code >= 500
 817                    else self.cooldown_seconds
 818                )
 819                rec.state = KeyState.COOLDOWN
 820                rec.cooldown_until = time.monotonic() + duration
 821                rec.failure_count += 1
 822                rec.last_error_code = err_code
 823                rec.last_error_message = err_msg
 824                callback = (self.on_cooldown, (mask(rec.key), duration, err_msg or err_code))
 825                result = KeyState.COOLDOWN
 826
 827        if callback is not None:
 828            self._invoke_callback(callback[0], *callback[1])
 829        if result == KeyState.COOLDOWN:
 830            logger.info("key %s entered cooldown for %.1fs", mask(key), callback[1][1] if callback else 0)
 831        elif result == KeyState.DISABLED:
 832            logger.info("key %s disabled", mask(key))
 833        return result

 835    def call(self, fn: Callable[..., Any], *args: Any, **kwargs: Any) -> Any:
 836        """Execute *fn* with an ACTIVE API key from the pool, handling rotation automatically."""
 837        last_exc: Exception | None = None
 838        started = time.monotonic()
 839        waited = 0.0
 840        attempts = 0
 841        while attempts < self.max_retries:
 842            try:
 843                key = self.get_key()
 844            except AllKeysExhaustedError as exc:
 845                remaining = self._seconds_until_soonest_recovery()
 846                budget = min(self.max_wait_seconds - (time.monotonic() - started), self.max_wait_seconds - waited)
 847                if self.wait_for_recovery and remaining is not None and budget > 0:
 848                    delay = min(remaining, budget)
 849                    logger.info("waiting %.1fs for key recovery", delay)
 850                    time.sleep(delay)
 851                    waited += delay
 852                    if waited >= self.max_wait_seconds:
 853                        break
 854                    continue
 855                self._invoke_callback(self.on_exhausted, self._safe_message(str(exc)))
 856                logger.warning("key pool exhausted")
 857                raise
 858            attempts += 1
 859            try:
 860                res = fn(key, *args, **kwargs)
 861            except Exception as exc:
 862                resp = getattr(exc, "response", None)
 863                if resp is not None and hasattr(resp, "status_code"):
 864                    status_code = resp.status_code
 865                    headers = dict(resp.headers) if hasattr(resp, "headers") else {}
 866                    body = None
 867                    if hasattr(resp, "json"):
 868                        try:
 869                            body = resp.json()
 870                        except Exception:
 871                            body = getattr(resp, "text", str(exc))
 872                    else:
 873                        body = getattr(resp, "text", str(exc))
 874                    state = self.handle_response(key, status_code, headers, body)
 875                    if state == KeyState.ACTIVE:
 876                        return resp
 877                    last_exc = exc
 878                    continue
 879                raise exc
 880
 881            if hasattr(res, "status_code"):
 882                headers = dict(res.headers) if hasattr(res, "headers") else {}
 883                body = None
 884                if hasattr(res, "json"):
 885                    try:
 886                        body = res.json()
 887                    except Exception:
 888                        body = getattr(res, "text", None)
 889                elif hasattr(res, "text"):
 890                    body = res.text
 891                state = self.handle_response(key, res.status_code, headers, body)
 892                if state == KeyState.ACTIVE:
 893                    return res
 894                continue
 895
 896            self.mark_success(key)
 897            return res
 898
 899        if last_exc:
 900            raise AllKeysExhaustedError(
 901                f"All retries ({self.max_retries}) exhausted. Last exception: {self._safe_message(str(last_exc))}"
 902            ) from last_exc
 903        raise AllKeysExhaustedError(
 904            f"All retries ({self.max_retries}) exhausted without a successful response."
 905        )
